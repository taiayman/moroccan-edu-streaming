// public/streaming/js/teacher.js
import { HMSSdk } from '@100mslive/hms-video';
import CONFIG from './config.js';

let hmsClient = null;
let localTracks = {
  audioTrack: null,
  videoTrack: null,
  screenTrack: null
};

async function initializeHMSClient() {
  try {
    hmsClient = new HMSSdk();
    
    const authToken = await getAuthToken(); // Implement token endpoint
    const config = {
      userName: 'Teacher',
      authToken,
      settings: {
        isAudioMuted: false,
        isVideoMuted: false,
      },
      rememberDeviceSelection: true,
    };

    await hmsClient.join(config);
    
    // Setup local tracks
    localTracks.audioTrack = await hmsClient.getLocalAudioTrack();
    localTracks.videoTrack = await hmsClient.getLocalVideoTrack();
    
    // Publish tracks
    await hmsClient.publish([localTracks.audioTrack, localTracks.videoTrack]);
    
    // Event listeners
    hmsClient.on('peer-joined', handlePeerJoined);
    hmsClient.on('peer-left', handlePeerLeft);
    hmsClient.on('track-updated', handleTrackUpdate);

  } catch (error) {
    console.error('HMS initialization failed:', error);
    showError('Failed to join room: ' + error.message);
  }
}

// Handle peer events
function handlePeerJoined(peer) {
  console.log('Peer joined:', peer.name);
  updateParticipantsList();
  
  if (peer.videoTrack) {
    const playerContainer = createRemotePlayerContainer(peer.id);
    hmsClient.attachVideoTrack(peer.videoTrack, playerContainer);
  }
}

function handlePeerLeft(peer) {
  console.log('Peer left:', peer.name);
  removeRemotePlayerContainer(peer.id);
  updateParticipantsList();
}

function handleTrackUpdate(track, peer) {
  console.log('Track updated:', track.type, peer.name);
  updateParticipantsList();
}

// Update media controls
async function toggleAudio() {
  if (localTracks.audioTrack) {
    await hmsClient.setLocalAudioEnabled(!isAudioEnabled);
    isAudioEnabled = !isAudioEnabled;
    updateUIState();
  }
}

async function toggleScreenShare() {
  if (!isScreenSharing) {
    localTracks.screenTrack = await hmsClient.getLocalScreenTrack();
    await hmsClient.publish([localTracks.screenTrack]);
    isScreenSharing = true;
  } else {
    await hmsClient.unpublish([localTracks.screenTrack]);
    isScreenSharing = false;
  }
}

const formatRelativeTime = (timestamp, t) => {
  if (!timestamp) return t('common.time.unknown');
  
  try {
    const now = new Date();
    const date = new Date(timestamp);
    if (isNaN(date.getTime())) return t('common.time.unknown');
    
    const diffInSeconds = Math.floor((now - date) / 1000);

    if (diffInSeconds < 60) {
      return t('common.time.justNow');
    }

    const diffInMinutes = Math.floor(diffInSeconds / 60);
    if (diffInMinutes < 60) {
      return t('common.time.minutesAgo', { count: diffInMinutes });
    }

    const diffInHours = Math.floor(diffInMinutes / 60);
    if (diffInHours < 24) {
      return t('common.time.hoursAgo', { count: diffInHours });
    }

    const diffInDays = Math.floor(diffInHours / 24);
    if (diffInDays < 7) {
      return t('common.time.daysAgo', { count: diffInDays });
    }

    return date.toLocaleDateString();
  } catch (error) {
    return t('common.time.unknown');
  }
};
